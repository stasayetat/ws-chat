# Backend WebSocket Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the NestJS WebSocket gateway with real-time messaging, in-memory state, and 4 bot users (Echo, Reverse, Spam, Ignore).

**Architecture:** A `ChatModule` owns a `ChatGateway` (handles WebSocket lifecycle and events) and a `ChatService` (all in-memory state). Bots are plain classes with a `handleMessage(msg, emit)` callback pattern — no NestJS coupling. The gateway instantiates each bot and wires their callbacks.

**Tech Stack:** NestJS 11, Socket.IO v4 via `@nestjs/websockets` + `@nestjs/platform-socket.io`, shared types from `@chat/api-interfaces`, Jest + `@nestjs/testing`

## WebSocket Event Contract

| Direction | Event | Payload |
|-----------|-------|---------|
| Client → Server | connection handshake | `{ auth: { name: string; avatar: string } }` |
| Client → Server | `sendMessage` | `SendMessageDto` |
| Server → Client | `contactsList` | `Contact[]` (sent once on connect) |
| Server → Client | `userConnected` | `User` (broadcast) |
| Server → Client | `userDisconnected` | `string` (userId, broadcast) |
| Server → Client | `newMessage` | `Message` (sent to receiver only) |

import { Injectable } from '@nestjs/common';
import { Contact, Message, User } from '@chat/api-interfaces';

const BOT_IDS = new Set(['bot-echo', 'bot-reverse', 'bot-spam', 'bot-ignore']);

const BOT_USERS: User[] = [
  { id: 'bot-echo', name: 'Echo Bot', avatar: '', status: 'online' },
  { id: 'bot-reverse', name: 'Reverse Bot', avatar: '', status: 'online' },
  { id: 'bot-spam', name: 'Spam Bot', avatar: '', status: 'online' },
  { id: 'bot-ignore', name: 'Ignore Bot', avatar: '', status: 'online' },
];

@Injectable()
export class ChatService {
  private readonly users = new Map<string, User>(
    BOT_USERS.map((b) => [b.id, b]),
  );
  // Key = [id1, id2].sort().join(':') so order does not matter
  private readonly messages = new Map<string, Message[]>();

  addUser(user: User): void {
    this.users.set(user.id, user);
  }

  removeUser(id: string): void {
    this.users.delete(id);
  }

  getUser(id: string): User | undefined {
    return this.users.get(id);
  }

  getContacts(excludeId: string): Contact[] {
    return [...this.users.values()]
      .filter((u) => u.id !== excludeId)
      .map((u) => ({ ...u }));
  }

  isBot(id: string): boolean {
    return BOT_IDS.has(id);
  }

  addMessage(message: Message): void {
    const key = this.conversationKey(message.senderId, message.receiverId);
    const history = this.messages.get(key) ?? [];
    history.push(message);
    this.messages.set(key, history);
  }

  getHistory(userId1: string, userId2: string): Message[] {
    return this.messages.get(this.conversationKey(userId1, userId2)) ?? [];
  }

  private conversationKey(a: string, b: string): string {
    return [a, b].sort().join(':');
  }
}
```

    setTimeout(() => {
      const reply: Message = {
        id: crypto.randomUUID(),
        senderId: this.id,
        receiverId: message.senderId,
        text: message.text.split('').reverse().join(''),
        timestamp: Date.now(),

      emit('newMessage', reply);
    }, 3000);

}

import { Message } from '@chat/api-interfaces';
import { Bot, EmitFn } from './bot.interface';

const PHRASES = [
  'Hello there!',
  'How is everyone doing?',
  'Anyone up for a chat?',
  'Nice to see you all!',
  'What a wonderful day!',
];

export class SpamBot implements Bot {
  readonly id = 'bot-spam';
  private timer: ReturnType<typeof setTimeout> | null = null;

  /** SpamBot ignores incoming messages */
  handleMessage(_message: Message, _emit: EmitFn): void {}

  /**
   * Starts the spam loop. Call once from the gateway's afterInit.
   * @param getRecipients - Returns current non-bot user IDs to spam
   * @param send - Callback to deliver a message to a specific user
   */
  startSpamming(
    getRecipients: () => string[],
    send: (receiverId: string, msg: Message) => void,
  ): void {
    this.scheduleNext(getRecipients, send);
  }

  stopSpamming(): void {
    if (this.timer) {
      clearTimeout(this.timer);
      this.timer = null;
    }
  }

  private scheduleNext(
    getRecipients: () => string[],
    send: (receiverId: string, msg: Message) => void,
  ): void {
    // Random delay between 10 and 120 seconds
    const delay = (10 + Math.random() * 110) * 1000;
    this.timer = setTimeout(() => {
      const recipients = getRecipients();
      if (recipients.length > 0) {
        const receiverId = recipients[Math.floor(Math.random() * recipients.length)];
        send(receiverId, {
          id: crypto.randomUUID(),
          senderId: this.id,
          receiverId,
          text: PHRASES[Math.floor(Math.random() * PHRASES.length)],
          timestamp: Date.now(),
        });
      }
      this.scheduleNext(getRecipients, send);
    }, delay);
  }
}
```

```typescript
import {
  ConnectedSocket,
  MessageBody,
  OnGatewayConnection,
  OnGatewayDisconnect,
  OnGatewayInit,
  SubscribeMessage,
  WebSocketGateway,
  WebSocketServer,
} from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';
import { Message, SendMessageDto, User } from '@chat/api-interfaces';
import { ChatService } from './chat.service';
import { Bot } from './bots/bot.interface';
import { EchoBot } from './bots/echo.bot';
import { ReverseBot } from './bots/reverse.bot';
import { SpamBot } from './bots/spam.bot';
import { IgnoreBot } from './bots/ignore.bot';

@WebSocketGateway({ cors: { origin: 'http://localhost:4200' } })
export class ChatGateway
  implements OnGatewayInit, OnGatewayConnection, OnGatewayDisconnect
{
  @WebSocketServer()
  private server!: Server;

  private readonly socketToUser = new Map<string, string>();
  private readonly userToSocket = new Map<string, string>();

  private readonly echoBot = new EchoBot();
  private readonly reverseBot = new ReverseBot();
  private readonly spamBot = new SpamBot();
  private readonly ignoreBot = new IgnoreBot();
  private readonly bots = new Map<string, Bot>([
    [this.echoBot.id, this.echoBot],
    [this.reverseBot.id, this.reverseBot],
    [this.spamBot.id, this.spamBot],
    [this.ignoreBot.id, this.ignoreBot],
  ]);

  constructor(private readonly chatService: ChatService) {}

  afterInit(_server: Server): void {
    this.spamBot.startSpamming(
      () =>
        [...this.userToSocket.keys()].filter(
          (id) => !this.chatService.isBot(id),
        ),
      (receiverId: string, msg: Message) => {
        const socketId = this.userToSocket.get(receiverId);
        if (socketId) {
          this.server.to(socketId).emit('newMessage', msg);
          this.chatService.addMessage(msg);
        }
      },
    );
  }

  handleConnection(client: Socket): void {
    const auth = client.handshake.auth as { name?: string; avatar?: string };
    if (!auth.name) {
      client.disconnect();
      return;
    }

    const user: User = {
      id: crypto.randomUUID(),
      name: auth.name,
      avatar: auth.avatar ?? '',
      status: 'online',
    };

    this.chatService.addUser(user);
    this.socketToUser.set(client.id, user.id);
    this.userToSocket.set(user.id, client.id);

    client.emit('contactsList', this.chatService.getContacts(user.id));
    client.broadcast.emit('userConnected', user);
  }

  handleDisconnect(client: Socket): void {
    const userId = this.socketToUser.get(client.id);
    if (!userId) return;

    this.chatService.removeUser(userId);
    this.socketToUser.delete(client.id);
    this.userToSocket.delete(userId);

    this.server?.emit('userDisconnected', userId);
  }

  @SubscribeMessage('sendMessage')
  handleSendMessage(
    @ConnectedSocket() client: Socket,
    @MessageBody() dto: SendMessageDto,
  ): void {
    const senderId = this.socketToUser.get(client.id);
    if (!senderId) return;

    const message: Message = {
      id: crypto.randomUUID(),
      senderId,
      receiverId: dto.receiverId,
      text: dto.text,
      timestamp: Date.now(),
    };

    this.chatService.addMessage(message);

    const bot = this.bots.get(dto.receiverId);
    if (bot) {
      // Route to bot — reply is sent back to the sender's socket
      bot.handleMessage(message, (event, data) => {
        const reply = data as Message;
        this.chatService.addMessage(reply);
        client.emit(event, reply);
      });
    } else {
      const receiverSocketId = this.userToSocket.get(dto.receiverId);
      if (receiverSocketId) {
        this.server.to(receiverSocketId).emit('newMessage', message);
      }
    }
  }
}
```
